def cholesky_portfolio_risk(weights, cov_matrix, alpha=0.99):
    # Decompose covariance matrix Σ = L * L.T
    L = np.linalg.cholesky(cov_matrix)
    portfolio_std = np.sqrt(weights.T @ (L @ L.T) @ weights)
    z_score = scipy.stats.norm.ppf(alpha)
    var_99 = z_score * portfolio_std
    return float(var_99)
